gyo_get_app_target("${GYO_CURRENT_APP}" client_network network)
gyo_get_app_target("${GYO_CURRENT_APP}" runtime_host host)
set(probe "gyo_${GYO_CURRENT_APP}_network_probe")
add_executable(${probe} network_main.cpp)
target_compile_features(${probe} PRIVATE cxx_std_20)
target_link_libraries(${probe} PRIVATE ${network} ${host} httplib::httplib "gyo_${GYO_CURRENT_APP}_client_v6")
gyo_register_acceptance_probe(OWNER "${GYO_CURRENT_APP}" ROLE network TARGET ${probe})
set(timing "gyo_${GYO_CURRENT_APP}_timing_probe")
add_executable(${timing} timing_main.cpp)
target_compile_features(${timing} PRIVATE cxx_std_20)
target_link_libraries(${timing} PRIVATE ${network})
gyo_register_acceptance_probe(OWNER "${GYO_CURRENT_APP}" ROLE timing TARGET ${timing})
set(action "gyo_${GYO_CURRENT_APP}_action_probe")

target_link_libraries(${action} PRIVATE ${network} nlohmann_json::nlohmann_json)
gyo_register_acceptance_probe(OWNER "${GYO_CURRENT_APP}" ROLE action TARGET ${action})
set(worker "gyo_${GYO_CURRENT_APP}_worker_probe")
add_executable(${worker} worker_main.cpp)
target_compile_features(${worker} PRIVATE cxx_std_20)
target_link_libraries(${worker} PRIVATE ${network} asio::asio httplib::httplib
    nlohmann_json::nlohmann_json "gyo_${GYO_CURRENT_APP}_client_v6")
gyo_register_acceptance_probe(OWNER "${GYO_CURRENT_APP}" ROLE worker TARGET ${worker})
