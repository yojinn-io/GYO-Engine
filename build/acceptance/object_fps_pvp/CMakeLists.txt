gyo_get_app_target("${GYO_CURRENT_APP}" client_network network)
gyo_get_app_target("${GYO_CURRENT_APP}" runtime_host host)
set(probe "gyo_${GYO_CURRENT_APP}_network_probe")
add_executable(${probe} network_main.cpp)
target_compile_features(${probe} PRIVATE cxx_std_20)
target_link_libraries(${probe} PRIVATE ${network} ${host} httplib::httplib)
gyo_register_acceptance_probe(OWNER "${GYO_CURRENT_APP}" ROLE network TARGET ${probe})
set(timing "gyo_${GYO_CURRENT_APP}_timing_probe")
add_executable(${timing} timing_main.cpp)
target_compile_features(${timing} PRIVATE cxx_std_20)
target_link_libraries(${timing} PRIVATE ${network})
gyo_register_acceptance_probe(OWNER "${GYO_CURRENT_APP}" ROLE timing TARGET ${timing})
set(worker "gyo_${GYO_CURRENT_APP}_worker_probe")
add_executable(${worker} worker_main.cpp)
target_compile_features(${worker} PRIVATE cxx_std_20)
target_link_libraries(${worker} PRIVATE ${network} asio::asio httplib::httplib
    nlohmann_json::nlohmann_json "gyo_${GYO_CURRENT_APP}_client_v3")
gyo_register_acceptance_probe(OWNER "${GYO_CURRENT_APP}" ROLE worker TARGET ${worker})
if(MSVC)
    target_compile_options(${probe} PRIVATE /utf-8)
endif()
gyo_get_app_target("${GYO_CURRENT_APP}" app_support support)
set(gui "gyo_${GYO_CURRENT_APP}_gui_probe")
add_executable(${gui} gui_main.cpp)
target_compile_features(${gui} PRIVATE cxx_std_20)
target_link_libraries(${gui} PRIVATE ${support} GYO::PlatformSDL GYO::RenderBackendSDLGPU)
gyo_register_acceptance_probe(OWNER "${GYO_CURRENT_APP}" ROLE gui TARGET ${gui})
if(MSVC)
    target_compile_options(${gui} PRIVATE /utf-8)
endif()
gyo_copy_runtime_dlls(${gui})
if(BUILD_TESTING)
    add_test(NAME "${GYO_CURRENT_APP}.worker" COMMAND ${worker})
    set_tests_properties("${GYO_CURRENT_APP}.worker" PROPERTIES LABELS "network;pvp" TIMEOUT 15)
    add_test(NAME "${GYO_CURRENT_APP}.wire" COMMAND ${probe} --self-test)
    set_tests_properties("${GYO_CURRENT_APP}.wire" PROPERTIES LABELS "cpu;pvp" TIMEOUT 10)
endif()
