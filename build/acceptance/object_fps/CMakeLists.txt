gyo_get_app_target("${GYO_CURRENT_APP}" app_support support)
set(probe "gyo_${GYO_CURRENT_APP}_acceptance")
add_executable(${probe} main.cpp diagnostics/HeadlessSmoke.cpp
    diagnostics/MuzzleProbe.cpp diagnostics/PackageProbes.cpp diagnostics/VisualProbe.cpp)
target_compile_features(${probe} PRIVATE cxx_std_20)
target_include_directories(${probe} PRIVATE "${GYO_APP_BINARY_DIR}/generated")
target_link_libraries(${probe} PRIVATE ${support} GYO::PlatformSDL GYO::RenderBackendSDLGPU)
set_target_properties(${probe} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/acceptance/${GYO_CURRENT_APP}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(${probe} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY_${config} "${PROJECT_BINARY_DIR}/acceptance/${GYO_CURRENT_APP}/bin")
endforeach()
if(WIN32)
    target_compile_definitions(${probe} PRIVATE NOMINMAX)
endif()
if(MSVC)
    target_compile_options(${probe} PRIVATE /utf-8)
endif()
gyo_copy_runtime_dlls(${probe})
gyo_stage_app_content(${probe} "${GYO_CURRENT_APP}" "${PROJECT_BINARY_DIR}/acceptance/${GYO_CURRENT_APP}")

if(BUILD_TESTING)
    foreach(mode startup headless)
        add_test(NAME "${GYO_CURRENT_APP}.${mode}_smoke" COMMAND ${probe} "--${mode}-smoke-test")
        set_tests_properties("${GYO_CURRENT_APP}.${mode}_smoke" PROPERTIES LABELS "cpu;smoke" TIMEOUT 90)
    endforeach()
    add_test(NAME "${GYO_CURRENT_APP}.package" COMMAND ${probe} --validate-package)
    set_tests_properties("${GYO_CURRENT_APP}.package" PROPERTIES LABELS "cpu;content" TIMEOUT 90)
    foreach(mode smoke menu_smoke viewmodel_smoke reload_smoke muzzle_smoke)
        string(REPLACE "_" "-" flag "${mode}")
        add_test(NAME "${GYO_CURRENT_APP}.${mode}" COMMAND ${probe} "--${flag}-test")
        set_tests_properties("${GYO_CURRENT_APP}.${mode}" PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
    endforeach()
    foreach(aspect 4x3 21x9)
        add_test(NAME "${GYO_CURRENT_APP}.muzzle_smoke_${aspect}"
            COMMAND ${probe} --muzzle-smoke-test "--preview-${aspect}")
        set_tests_properties("${GYO_CURRENT_APP}.muzzle_smoke_${aspect}"
            PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
    endforeach()
endif()
