gyo_get_app_target("${GYO_CURRENT_APP}" app_support support)
set(probe "gyo_${GYO_CURRENT_APP}_acceptance")
add_executable(${probe} main.cpp RigInspection.cpp)
target_compile_features(${probe} PRIVATE cxx_std_20)
target_link_libraries(${probe} PRIVATE ${support} GYO::PlatformSDL
    GYO::RenderBackendSDLGPU GYO::AssetUfbx)
gyo_use_app_config(${probe} "${GYO_CURRENT_APP}")
gyo_register_acceptance_probe(OWNER "${GYO_CURRENT_APP}" ROLE main TARGET ${probe})
set(stage "${PROJECT_BINARY_DIR}/acceptance/${GYO_CURRENT_APP}")
set_target_properties(${probe} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${stage}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(${probe} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${stage}/bin")
endforeach()
if(WIN32)
    target_compile_definitions(${probe} PRIVATE NOMINMAX)
endif()
if(MSVC)
    target_compile_options(${probe} PRIVATE /utf-8)
endif()
gyo_copy_runtime_dlls(${probe})
gyo_stage_app_content(${probe} "${GYO_CURRENT_APP}" "${stage}")
if(BUILD_TESTING)
    foreach(mode startup headless)
        add_test(NAME "${GYO_CURRENT_APP}.${mode}_smoke" COMMAND ${probe} "--${mode}-smoke-test")
        set_tests_properties("${GYO_CURRENT_APP}.${mode}_smoke" PROPERTIES LABELS "cpu;smoke" TIMEOUT 120)
    endforeach()
    add_test(NAME "${GYO_CURRENT_APP}.enemy_gpu" COMMAND ${probe} --smoke-test)
    set_tests_properties("${GYO_CURRENT_APP}.enemy_gpu" PROPERTIES LABELS gpu TIMEOUT 180 RUN_SERIAL TRUE)
endif()
