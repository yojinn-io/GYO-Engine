gyo_get_app_target("${GYO_CURRENT_APP}" app_support support)
set(test_stage "${PROJECT_BINARY_DIR}/test-products/${GYO_CURRENT_APP}")
set(test_content "gyo_${GYO_CURRENT_APP}_test_runtime")
add_custom_target(${test_content})
gyo_stage_app_content(${test_content} "${GYO_CURRENT_APP}" "${test_stage}")
set(tests "gyo_${GYO_CURRENT_APP}_tests")
add_executable(${tests} TestMain.cpp AssetIntegrationTests.cpp PresentationTests.cpp
    EnemySystemTests.cpp EnemyAttachmentTests.cpp CharacterAccessoryTests.cpp RegionShotTests.cpp
    App/DebugUiTests.cpp App/DebugRuntimeTests.cpp)
target_compile_features(${tests} PRIVATE cxx_std_20)
target_link_libraries(${tests} PRIVATE ${support} GYO::AssetUfbx GYO::AssetSdlImage
    GYO::PlatformSDL GYO::InputBackendSDL GYO::UiRenderer doctest::doctest)
target_include_directories(${tests} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}"
    "${PROJECT_SOURCE_DIR}/tests/common/support")
gyo_use_app_config(${tests} "${GYO_CURRENT_APP}")
set_target_properties(${tests} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${test_stage}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(${tests} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${test_stage}/bin")
endforeach()
if(WIN32)
    target_compile_definitions(${tests} PRIVATE NOMINMAX)
endif()
if(MSVC)
    target_compile_options(${tests} PRIVATE /utf-8)
endif()
gyo_copy_runtime_dlls(${tests})
add_dependencies(${tests} ${test_content})
add_test(NAME "${GYO_CURRENT_APP}.cpu" COMMAND ${tests})
set_tests_properties("${GYO_CURRENT_APP}.cpu" PROPERTIES LABELS "cpu;gameplay" TIMEOUT 180
    ENVIRONMENT "SDL_VIDEODRIVER=dummy")
