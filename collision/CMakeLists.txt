add_library(gyo_collision STATIC src/Collision.cpp)
add_library(GYO::Collision ALIAS gyo_collision)
target_include_directories(gyo_collision PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(gyo_collision PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(gyo_collision PRIVATE /W4 /permissive-)
    target_compile_definitions(gyo_collision PRIVATE NOMINMAX)
else()
    target_compile_options(gyo_collision PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
    add_executable(gyo_collision_tests tests/CollisionTests.cpp)
    target_link_libraries(gyo_collision_tests PRIVATE GYO::Collision doctest::doctest)
    add_test(NAME gyo_collision_tests COMMAND gyo_collision_tests)
endif()
